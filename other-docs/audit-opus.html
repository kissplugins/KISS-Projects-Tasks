🛡️ FSM Guard Rails & Recovery System

Immediate fixes to stabilize your Timer/Session FSM implementation

🚨 Problem Analysis

Your FSM system is brittle due to:

  • State Synchronization Issues: TimerFSM and SessionFSM aren't properly synchronized
  • Stuck States: SessionFSM gets stuck in EDITING state with no recovery mechanism
  • Event Handler Conflicts: Multiple event handlers competing for the same actions
  • Missing State Validation: No guards preventing invalid state transitions
  • Poor Error Recovery: When things go wrong, the FSMs don't self-heal

🛠️ Immediate Guard Rails Implementation

File: assets/js/fsm/guards/FSMGuardSystem.js
// Create this new file for centralized guard system
(function(root) {
    'use strict';
    
    /**
     * FSM Guard System - Prevents invalid states and provides recovery
     */
    function FSMGuardSystem() {
        this.watchdogInterval = null;
        this.stateHistory = [];
        this.maxHistorySize = 50;
        this.stuckStateThreshold = 5000; // 5 seconds
        this.lastStateChange = {};
    }
    
    /**
     * Initialize the guard system for both FSMs
     */
    FSMGuardSystem.prototype.init = function() {
        if (!root.PTT_FSM_ENABLED) return;
        
        this.startWatchdog();
        this.installStateInterceptors();
        this.installEventValidation();
        this.installRecoveryMechanisms();
        
        console.log('[FSM Guard] System initialized');
    };
    
    /**
     * Watchdog timer to detect stuck states
     */
    FSMGuardSystem.prototype.startWatchdog = function() {
        var self = this;
        
        this.watchdogInterval = setInterval(function() {
            self.checkForStuckStates();
            self.validateFSMConsistency();
            self.cleanupOrphanedState();
        }, 2000); // Check every 2 seconds
    };
    
    /**
     * Check if any FSM is stuck in a state too long
     */
    FSMGuardSystem.prototype.checkForStuckStates = function() {
        var now = Date.now();
        
        // Check SessionFSM
        if (root.PTT_SessionFSM) {
            var sessionState = root.PTT_SessionFSM.state;
            var lastChange = this.lastStateChange['session_' + sessionState] || now;
            
            if (sessionState === 'EDITING' && (now - lastChange) > this.stuckStateThreshold) {
                if (!root.PTT_SessionFSM._hasMeaningfulChanges()) {
                    console.warn('[FSM Guard] SessionFSM stuck in EDITING with no changes - recovering');
                    root.PTT_SessionFSM.transition('RESET');
                }
            }
            
            if (sessionState === 'SAVING' && (now - lastChange) > 30000) {
                console.error('[FSM Guard] SessionFSM stuck in SAVING for 30s - force reset');
                root.PTT_SessionFSM.forceReset();
            }
        }
        
        // Check TimerFSM
        if (root.PTT_EditorFSM) {
            var timerState = root.PTT_EditorFSM.state;
            var lastChange = this.lastStateChange['timer_' + timerState] || now;
            
            if (timerState === 'STARTING' && (now - lastChange) > 10000) {
                console.error('[FSM Guard] TimerFSM stuck in STARTING for 10s - resetting');
                root.PTT_EditorFSM.state = 'IDLE';
                root.PTT_EditorFSM.ctx = { taskId: null, postId: null, sessionIndex: null, startUtc: null };
            }
        }
    };
    
    /**
     * Validate consistency between Timer and Session FSMs
     */
    FSMGuardSystem.prototype.validateFSMConsistency = function() {
        if (!root.PTT_EditorFSM || !root.PTT_SessionFSM) return;
        
        var timerState = root.PTT_EditorFSM.state;
        var sessionState = root.PTT_SessionFSM.state;
        
        // Invalid state combinations
        var invalidCombos = [
            { timer: 'RUNNING', session: 'CREATING', action: 'stop_timer' },
            { timer: 'IDLE', session: 'AUTO_SAVING', action: 'reset_session' },
            { timer: 'RUNNING', session: 'DELETING', action: 'stop_timer' }
        ];
        
        invalidCombos.forEach(function(combo) {
            if (timerState === combo.timer && sessionState === combo.session) {
                console.warn('[FSM Guard] Invalid state combination detected:', combo);
                if (combo.action === 'stop_timer') {
                    root.PTT_EditorFSM.transition('STOP_TIMER');
                } else if (combo.action === 'reset_session') {
                    root.PTT_SessionFSM.transition('RESET');
                }
            }
        });
    };
    
    /**
     * Clean up orphaned state data
     */
    FSMGuardSystem.prototype.cleanupOrphanedState = function() {
        // Clean up session storage flags
        var flags = [
            'ptt_session_save_pending',
            'ptt_session_delete_pending',
            'ptt_session_duplicate_pending',
            'ptt_session_reorder_pending',
            'ptt_session_bulk_delete_pending',
            'ptt_session_autosave_pending'
        ];
        
        flags.forEach(function(flag) {
            var value = sessionStorage.getItem(flag);
            if (value === '1') {
                var key = 'ptt_guard_' + flag + '_time';
                var setTime = sessionStorage.getItem(key);
                var now = Date.now();
                
                if (!setTime) {
                    sessionStorage.setItem(key, now);
                } else if (now - parseInt(setTime) > 30000) {
                    // Flag has been set for more than 30 seconds - clear it
                    console.warn('[FSM Guard] Clearing orphaned flag:', flag);
                    sessionStorage.removeItem(flag);
                    sessionStorage.removeItem(key);
                }
            }
        });
    };
    
    /**
     * Install state change interceptors
     */
    FSMGuardSystem.prototype.installStateInterceptors = function() {
        var self = this;
        
        // Intercept SessionFSM transitions
        if (root.PTT_SessionFSM) {
            var originalTransition = root.PTT_SessionFSM.transition;
            root.PTT_SessionFSM.transition = function(event, payload) {
                var prevState = this.state;
                
                // Pre-transition validation
                if (!self.validateTransition('session', prevState, event)) {
                    console.warn('[FSM Guard] Blocked invalid transition:', prevState, '->', event);
                    return;
                }
                
                // Call original transition
                var result = originalTransition.call(this, event, payload);
                
                // Record state change
                self.recordStateChange('session', prevState, this.state, event);
                
                return result;
            };
        }
        
        // Intercept TimerFSM transitions
        if (root.PTT_EditorFSM) {
            var originalTransition = root.PTT_EditorFSM.transition;
            root.PTT_EditorFSM.transition = function(event, payload) {
                var prevState = this.state;
                
                // Pre-transition validation
                if (!self.validateTransition('timer', prevState, event)) {
                    console.warn('[FSM Guard] Blocked invalid transition:', prevState, '->', event);
                    return;
                }
                
                // Call original transition
                var result = originalTransition.call(this, event, payload);
                
                // Record state change
                self.recordStateChange('timer', prevState, this.state, event);
                
                return result;
            };
        }
    };
    
    /**
     * Validate if a transition is allowed
     */
    FSMGuardSystem.prototype.validateTransition = function(fsm, state, event) {
        // Define invalid transitions
        var invalidTransitions = {
            session: {
                'SAVING': ['CREATE_SESSION', 'EDIT_SESSION', 'DELETE_SESSION'],
                'DELETING': ['CREATE_SESSION', 'EDIT_SESSION', 'SAVE_SESSION'],
                'AUTO_SAVING': ['CREATE_SESSION', 'DELETE_SESSION']
            },
            timer: {
                'STARTING': ['START_TIMER'],
                'STOPPING': ['STOP_TIMER'],
                'RUNNING': ['START_TIMER']
            }
        };
        
        var fsmInvalid = invalidTransitions[fsm];
        if (!fsmInvalid) return true;
        
        var stateInvalid = fsmInvalid[state];
        if (!stateInvalid) return true;
        
        return stateInvalid.indexOf(event) === -1;
    };
    
    /**
     * Record state changes for history and debugging
     */
    FSMGuardSystem.prototype.recordStateChange = function(fsm, prevState, newState, event) {
        var now = Date.now();
        
        this.lastStateChange[fsm + '_' + newState] = now;
        
        this.stateHistory.push({
            fsm: fsm,
            prev: prevState,
            next: newState,
            event: event,
            timestamp: now
        });
        
        // Keep history size limited
        if (this.stateHistory.length > this.maxHistorySize) {
            this.stateHistory.shift();
        }
    };
    
    /**
     * Install event validation to prevent duplicate handlers
     */
    FSMGuardSystem.prototype.installEventValidation = function() {
        var self = this;
        var processedEvents = {};
        
        // Override jQuery event binding to detect duplicates
        var originalOn = jQuery.fn.on;
        jQuery.fn.on = function(events, selector, data, handler) {
            if (typeof events === 'string' && events.indexOf('ptt') !== -1) {
                var eventKey = events + '_' + (selector || 'document');
                
                if (processedEvents[eventKey]) {
                    console.warn('[FSM Guard] Duplicate event handler detected:', eventKey);
                    // Allow but warn - don't block in case it's intentional
                }
                processedEvents[eventKey] = true;
            }
            
            return originalOn.apply(this, arguments);
        };
    };
    
    /**
     * Install recovery mechanisms
     */
    FSMGuardSystem.prototype.installRecoveryMechanisms = function() {
        var self = this;
        
        // Global error handler for FSM errors
        window.addEventListener('error', function(e) {
            if (e.message && e.message.indexOf('FSM') !== -1) {
                console.error('[FSM Guard] Caught FSM error:', e.message);
                self.attemptRecovery();
            }
        });
        
        // Add keyboard shortcut for emergency reset (Ctrl+Shift+R)
        document.addEventListener('keydown', function(e) {
            if (e.ctrlKey && e.shiftKey && e.key === 'R') {
                e.preventDefault();
                if (confirm('Reset all FSM states? This will clear any pending operations.')) {
                    self.emergencyReset();
                }
            }
        });
    };
    
    /**
     * Attempt automatic recovery
     */
    FSMGuardSystem.prototype.attemptRecovery = function() {
        console.warn('[FSM Guard] Attempting automatic recovery...');
        
        // Reset stuck FSMs
        if (root.PTT_SessionFSM && root.PTT_SessionFSM.state === 'ERROR') {
            root.PTT_SessionFSM.transition('RESET');
        }
        
        if (root.PTT_EditorFSM && root.PTT_EditorFSM.state === 'ERROR') {
            root.PTT_EditorFSM.state = 'IDLE';
        }
        
        // Clear all pending flags
        this.cleanupOrphanedState();
        
        // Re-enable UI elements
        jQuery('.acf-button[disabled]').prop('disabled', false);
        jQuery('.ptt-session-start[disabled]').prop('disabled', false);
        
        console.log('[FSM Guard] Recovery attempt completed');
    };
    
    /**
     * Emergency reset - nuclear option
     */
    FSMGuardSystem.prototype.emergencyReset = function() {
        console.warn('[FSM Guard] EMERGENCY RESET INITIATED');
        
        // Reset SessionFSM
        if (root.PTT_SessionFSM) {
            root.PTT_SessionFSM.state = 'IDLE';
            root.PTT_SessionFSM.ctx = {
                sessionIndex: null,
                postId: null,
                isDirty: false,
                validationErrors: [],
                pendingChanges: {}
            };
        }
        
        // Reset TimerFSM
        if (root.PTT_EditorFSM) {
            root.PTT_EditorFSM.state = 'IDLE';
            root.PTT_EditorFSM.ctx = {
                taskId: null,
                postId: null,
                sessionIndex: null,
                startUtc: null
            };
        }
        
        // Clear all storage
        Object.keys(sessionStorage).forEach(function(key) {
            if (key.indexOf('ptt_') === 0) {
                sessionStorage.removeItem(key);
            }
        });
        
        // Reset UI
        location.reload();
    };
    
    /**
     * Get diagnostic information
     */
    FSMGuardSystem.prototype.getDiagnostics = function() {
        return {
            sessionFSM: root.PTT_SessionFSM ? {
                state: root.PTT_SessionFSM.state,
                context: root.PTT_SessionFSM.ctx,
                hasPendingChanges: root.PTT_SessionFSM.hasUnsavedChanges()
            } : null,
            timerFSM: root.PTT_EditorFSM ? {
                state: root.PTT_EditorFSM.state,
                context: root.PTT_EditorFSM.ctx
            } : null,
            stateHistory: this.stateHistory.slice(-10),
            orphanedFlags: this.getOrphanedFlags()
        };
    };
    
    /**
     * Get orphaned flags from session storage
     */
    FSMGuardSystem.prototype.getOrphanedFlags = function() {
        var orphaned = [];
        Object.keys(sessionStorage).forEach(function(key) {
            if (key.indexOf('ptt_session_') === 0 && key.indexOf('_pending') !== -1) {
                orphaned.push(key);
            }
        });
        return orphaned;
    };
    
    // Export to global scope
    root.PTT_FSMGuard = new FSMGuardSystem();
    
    // Auto-initialize when DOM is ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            root.PTT_FSMGuard.init();
        });
    } else {
        root.PTT_FSMGuard.init();
    }
    
    // Add console commands for debugging
    root.PTT_FSM_DEBUG = {
        diagnostics: function() {
            return root.PTT_FSMGuard.getDiagnostics();
        },
        reset: function() {
            root.PTT_FSMGuard.emergencyReset();
        },
        recover: function() {
            root.PTT_FSMGuard.attemptRecovery();
        },
        history: function() {
            return root.PTT_FSMGuard.stateHistory;
        }
    };
    
    console.log('[FSM Guard] Debug commands available: PTT_FSM_DEBUG.diagnostics(), .reset(), .recover(), .history()');
    
})(window);

SessionFSM.prototype.checkAndRecover = function() {
    if (this.state === 'EDITING' && !this._hasMeaningfulChanges()) {
        var now = Date.now();
        this._lastEditCheck = this._lastEditCheck || now;
        
        if (now - this._lastEditCheck > 3000) {
            console.log('[SessionFSM] Auto-recovering from stuck EDITING state');
            this.transition('RESET');
        }
    } else {
        this._lastEditCheck = Date.now();
    }
};
